source /data/pjh_workspace/ts-env/bin/activate
export CUDA_VISIBLE_DEVICES=1

ETTH1_S1_96="checkpoints/soft_set_mse/stage1/ETTh1/seq96_pred96/stage1_carts_softset_ETTh1_96_S0_wce_RelationStage1_ETTh1_ftM_sl96_ll0_pl96_dm128_nh4_el2_dl1_df256_expand2_dc4_fc1_ebtimeF_dtTrue_softset_S0_wce_ETTh1_sl96_pl96_0/checkpoint.pth"

# T1: SAME R2 hybrid loss as C0 (SmoothL1 + pairwise, lambda_rank=1.0),
# frozen encoder, cosine UtilityHead -- only the training-time PREFIX
# source changes from the Oracle's sequence to the model's own on-policy
# picks. No Oracle-Choice CE (D1's own change), no encoder training.
python -u scripts/train_onpolicy_prefix01.py \
  --base_ckpt "$ETTH1_S1_96" --teacher_cache "$TEACHER" \
  --checkpoints checkpoints/exp_onpolicy_prefix01 \
  --model_id carts_t1_ETTh1_H96 --des t1_ETTh1_H96 \
  --top_k 10 --seed 0 --chunk_size 4096 --lambda_rank 1.0 --keep_all_checkpoints

T1_CKPT="checkpoints/exp_onpolicy_prefix01/onpolicy_prefix01/ETTh1/seq96_pred96/carts_t1_ETTh1_H96/checkpoint.pth"

python -u scripts/eval_margutil01_stage2.py \
  --stage2_checkpoint "$ETTH1_S2_96" --sequential_checkpoint "$T1_CKPT" \
  --teacher_cache "$TEACHER" --top_k 10 --chunk_size 2048 --utility_eval_queries 200 \
  --out results/EXP-ONPOLICY-PREFIX01/ETTh1_H96_T1_stage2.json

  python -u scripts/eval_firstanchor_diag.py \
    --stage2_checkpoint "$ETTH1_S2_96" --sequential_checkpoint "$T1_CKPT" \
    --teacher_cache "$TEACHER" --top_k 10 --rank_eval_queries 200 --split "$SPLIT" \
    --out_dir results/EXP-ONPOLICY-PREFIX01 --cell_name "ETTh1_H96_T1_${SPLIT}"

  python -u scripts/eval_continuation_diag.py \
    --stage2_checkpoint "$ETTH1_S2_96" --sequential_checkpoint "$T1_CKPT" \
    --top_k 10 --query_budget 500 --chunk_size 2048 --split "$SPLIT" \
    --out_dir results/EXP-ONPOLICY-PREFIX01 --cell_name "ETTh1_H96_T1_${SPLIT}" --anchor_policy dense_first
done
