ETTH1_S1_96="checkpoints/soft_set_mse/stage1/ETTh1/seq96_pred96/stage1_carts_softset_ETTh1_96_S0_wce_RelationStage1_ETTh1_ftM_sl96_ll0_pl96_dm128_nh4_el2_dl1_df256_expand2_dc4_fc1_ebtimeF_dtTrue_softset_S0_wce_ETTh1_sl96_pl96_0/checkpoint.pth"

# E1: R2 hybrid loss + cosine UtilityHead (unchanged) + TRAINABLE encoder,
# initialised from the SAME B0 checkpoint. Only new flags vs. C0/R2: none --
# encoder freeze/unfreeze is a code-path change (requires_grad flip), not a
# CLI flag, since this experiment's whole point is that ONLY this changes.
python -u scripts/train_encoder_unfreeze01.py \
  --base_ckpt "$ETTH1_S1_96" --teacher_cache "$TEACHER" \
  --checkpoints checkpoints/exp_encoder_unfreeze01 \
  --model_id carts_e1_ETTh1_H96 --des e1_ETTh1_H96 \
  --top_k 10 --seed 0 --chunk_size 4096 --cand_chunk_size 2048 --lambda_rank 1.0 \
  --keep_all_checkpoints --probe_out results/EXP-ENCODER-UNFREEZE01/encoder_drift.csv

E1_CKPT="checkpoints/exp_encoder_unfreeze01/encoder_unfreeze01/ETTh1/seq96_pred96/carts_e1_ETTh1_H96/checkpoint.pth"

python -u scripts/eval_margutil01_stage2.py \
  --stage2_checkpoint "$ETTH1_S2_96" --sequential_checkpoint "$E1_CKPT" \
  --teacher_cache "$TEACHER" --top_k 10 --chunk_size 2048 --utility_eval_queries 200 \
  --out results/EXP-ENCODER-UNFREEZE01/ETTh1_H96_E1_stage2.json

# t=1 / t=2 diagnostics, BOTH test and train split (--split, added in
# EXP-STRONG-SCORER-DIAG01, reused unmodified here):
for SPLIT in test train; do
  python -u scripts/eval_firstanchor_diag.py \
    --stage2_checkpoint "$ETTH1_S2_96" --sequential_checkpoint "$E1_CKPT" \
    --teacher_cache "$TEACHER" --top_k 10 --rank_eval_queries 200 --split "$SPLIT" \
    --out_dir results/EXP-ENCODER-UNFREEZE01 --cell_name "ETTh1_H96_E1_${SPLIT}"

  python -u scripts/eval_continuation_diag.py \
    --stage2_checkpoint "$ETTH1_S2_96" --sequential_checkpoint "$E1_CKPT" \
    --top_k 10 --query_budget 500 --chunk_size 2048 --split "$SPLIT" \
    --out_dir results/EXP-ENCODER-UNFREEZE01 --cell_name "ETTh1_H96_E1_${SPLIT}" --anchor_policy dense_first
done
