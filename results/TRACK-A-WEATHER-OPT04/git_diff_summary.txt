+def resolve_oracle_compute_impl(requested_impl, chunk_size):

+    fallback_reason: {choice_ce, individual, greedy_set}}.
+    """
+    if requested_impl not in ORACLE_COMPUTE_IMPLS:
+        raise ValueError(f'--oracle_compute_impl must be one of {ORACLE_COMPUTE_IMPLS}, '
+                         f'got {requested_impl!r}')
+    reasons = {'choice_ce': '', 'individual': '', 'greedy_set': ''}
+    if requested_impl == 'reference':
+        reasons = {k: 'requested_impl=reference (default)' for k in reasons}
+        return {'choice_ce_impl': 'reference', 'individual_impl': 'reference',
+               'greedy_set_impl': 'reference', 'fallback_reason': reasons}
+
+    # requested_impl == 'optimized'
+    choice_ce_impl = 'optimized'         # C: unconditional (OPT03/OPT04 finding)
+    individual_impl = 'optimized_cache'  # A: unconditional CACHING of the reference

+                                         # norm-expansion algebra, which stays

+    if chunk_size >= GREEDY_SET_OPTIMIZED_MIN_CHUNK_SIZE:
+        greedy_set_impl = 'optimized'
+    else:
