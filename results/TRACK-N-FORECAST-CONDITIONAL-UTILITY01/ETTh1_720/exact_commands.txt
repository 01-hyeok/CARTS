  --shared_gate_init_out logs/TRACK-N-FORECAST-CONDITIONAL-UTILITY01/shared_gate_init_ETTh1_720.pth

    --cache_dir "results/TRACK-N-FORECAST-CONDITIONAL-UTILITY01/gate_only/cache/ETTh1_720/$ARM" \
    --seed 0 --train_epochs 10 --patience 5 \
    --shared_gate_init_in logs/TRACK-N-FORECAST-CONDITIONAL-UTILITY01/shared_gate_init_ETTh1_720.pth
